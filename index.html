/* ========================================================= 🔊 NEGIN AAC PROFESSIONAL PERSIAN TEXT-TO-SPEECH ENGINE ========================================================= */ let availableVoices = []; let selectedPersianVoice = null; /* ========================================================= بررسی پشتیبانی مرورگر ========================================================= */ function speechSupported(){ return ( "speechSynthesis" in window && "SpeechSynthesisUtterance" in window ); } /* ========================================================= دریافت صداهای دستگاه ========================================================= */ function loadVoices(){ if(!speechSupported()){ availableVoices = []; selectedPersianVoice = null; return; } try{ availableVoices = window.speechSynthesis.getVoices() || []; /* اولویت انتخاب صدا: 1. فارسی افغانستان fa-AF 2. فارسی ایران fa-IR 3. فارسی عمومی fa 4. هر صدای فارسی fa-* */ selectedPersianVoice = availableVoices.find( voice => voice.lang && voice.lang.toLowerCase() === "fa-af" ) || availableVoices.find( voice => voice.lang && voice.lang.toLowerCase() === "fa-ir" ) || availableVoices.find( voice => voice.lang && voice.lang.toLowerCase() === "fa" ) || availableVoices.find( voice => voice.lang && voice.lang .toLowerCase() .startsWith("fa-") ) || null; console.log( "Negin AAC - Available voices:", availableVoices ); console.log( "Negin AAC - Persian voice:", selectedPersianVoice ); }catch(error){ console.error( "Negin AAC voice loading error:", error ); availableVoices = []; selectedPersianVoice = null; } } /* ========================================================= بارگذاری اولیه صداها ========================================================= */ if(speechSupported()){ loadVoices(); /* بعضی مرورگرها مخصوصاً Chrome صداها را بعد از بارگذاری صفحه ارائه می‌کنند. */ if( typeof window.speechSynthesis .addEventListener === "function" ){ window.speechSynthesis.addEventListener( "voiceschanged", loadVoices ); } } /* ========================================================= دریافت بهترین صدای فارسی ========================================================= */ function getPersianVoice(){ loadVoices(); return selectedPersianVoice; } /* ========================================================= دریافت وضعیت صدای فارسی ========================================================= */ function hasPersianVoice(){ const voice = getPersianVoice(); return !!voice; } /* ========================================================= نمایش وضعیت صدا ========================================================= */ function getSpeechStatus(){ if(!speechSupported()){ return { supported: false, voice: null, message: "مرورگر شما از تبدیل متن به صدا پشتیبانی نمی‌کند." }; } const voice = getPersianVoice(); if(voice){ return { supported: true, voice: voice, message: "صدای فارسی آماده است." }; } return { supported: true, voice: null, message: "صدای فارسی روی دستگاه پیدا نشد." }; } /* ========================================================= 🔊 خواندن متن ========================================================= */ function speak(text){ /* تبدیل مقدار به متن */ text = String(text || "").trim(); /* اگر متن خالی باشد */ if(!text){ if(typeof showToast === "function"){ showToast( "ابتدا یک جمله بسازید." ); } return; } /* بررسی پشتیبانی */ if(!speechSupported()){ if(typeof showToast === "function"){ showToast( "تبدیل متن به صدا در این مرورگر پشتیبانی نمی‌شود." ); } return; } const synth = window.speechSynthesis; /* متوقف کردن صدای قبلی */ try{ synth.cancel(); }catch(error){ console.warn( "Negin AAC speech cancel error:", error ); } /* دریافت صدای فارسی */ const voice = getPersianVoice(); /* ساخت Utterance */ const utterance = new SpeechSynthesisUtterance(text); /* اگر صدای فارسی موجود باشد، همان صدا را مستقیماً انتخاب می‌کنیم. */ if(voice){ utterance.voice = voice; utterance.lang = voice.lang || "fa-IR"; }else{ /* اگر صدای fa-AF موجود نبود، fa-IR را به عنوان fallback استفاده می‌کنیم. */ utterance.lang = "fa-IR"; } /* سرعت خواندن */ utterance.rate = Number(speechRate) || 0.85; /* زیر و بمی صدا */ utterance.pitch = 1; /* حجم صدا */ utterance.volume = 1; /* ذخیره آخرین جمله */ lastSpokenText = text; /* ===================================================== شروع صدا ===================================================== */ utterance.onstart = function(){ console.log( "🔊 Negin AAC speech started:", text ); }; /* ===================================================== پایان صدا ===================================================== */ utterance.onend = function(){ console.log( "🔊 Negin AAC speech finished." ); }; /* ===================================================== توقف صدا ===================================================== */ utterance.onpause = function(){ console.log( "⏸ Negin AAC speech paused." ); }; /* ===================================================== ادامه صدا ===================================================== */ utterance.onresume = function(){ console.log( "▶️ Negin AAC speech resumed." ); }; /* ===================================================== خطای صدا ===================================================== */ utterance.onerror = function(event){ console.error( "❌ Negin AAC speech error:", event ); /* خطای اجازه */ if( event.error === "not-allowed" ){ if(typeof showToast === "function"){ showToast( "اجازه پخش صدا داده نشده است." ); } return; } /* صدای موجود نیست */ if( event.error === "voice-unavailable" ){ if(typeof showToast === "function"){ showToast( "صدای فارسی روی دستگاه پیدا نشد." ); } return; } /* خطای عمومی */ if(typeof showToast === "function"){ showToast( "پخش صدا انجام نشد. تنظیمات صدای دستگاه را بررسی کنید." ); } }; /* ===================================================== اجرای صدا ===================================================== */ try{ synth.speak( utterance ); /* بعضی دستگاه‌های Android ممکن است speech engine را در حالت paused نگه دارند. */ setTimeout( function(){ try{ if(synth.paused){ synth.resume(); } }catch(error){ console.warn( "Negin AAC speech resume error:", error ); } }, 150 ); }catch(error){ console.error( "❌ Negin AAC TTS error:", error ); if(typeof showToast === "function"){ showToast( "خطا در اجرای صدای فارسی." ); } } } /* ========================================================= 🔊 تست صدای فارسی ========================================================= */ function testPersianVoice(){ const testText = "سلام. من نگین هستم. آماده‌ام تا به شما کمک کنم."; speak( testText ); } /* ========================================================= 🔊 تست جمله دلخواه ========================================================= */ function testCustomVoice(text){ text = String(text || "").trim(); if(!text){ if(typeof showToast === "function"){ showToast( "متنی برای آزمایش وارد کنید." ); } return; } speak( text ); } /* ========================================================= ⏹ توقف صدا ========================================================= */ function stopSpeech(){ if(!speechSupported()){ return; } try{ window.speechSynthesis.cancel(); console.log( "⏹ Negin AAC speech stopped." ); }catch(error){ console.error( "Negin AAC stop speech error:", error ); } } /* ========================================================= ⏸ مکث صدا ========================================================= */ function pauseSpeech(){ if(!speechSupported()){ return; } try{ if( window.speechSynthesis.speaking ){ window.speechSynthesis.pause(); } }catch(error){ console.error( "Negin AAC pause speech error:", error ); } } /* ========================================================= ▶️ ادامه صدا ========================================================= */ function resumeSpeech(){ if(!speechSupported()){ return; } try{ window.speechSynthesis.resume(); }catch(error){ console.error( "Negin AAC resume speech error:", error ); } } /* ========================================================= 🔁 تکرار آخرین جمله ========================================================= */ function repeatSpeech(){ /* اگر آخرین جمله وجود داشته باشد */ if( lastSpokenText && lastSpokenText.trim() ){ speak( lastSpokenText ); return; } /* اگر جمله قبلی وجود ندارد، جمله فعلی را دریافت می‌کنیم. */ try{ const sentence = getSentence(); if(sentence){ speak( sentence ); }else{ if(typeof showToast === "function"){ showToast( "هنوز جمله‌ای برای تکرار وجود ندارد." ); } } }catch(error){ console.error( "Negin AAC repeat speech error:", error ); } } /* ========================================================= 🔊 خواندن جمله فعلی ========================================================= */ function speakCurrentSentence(){ try{ const sentence = getSentence(); if(!sentence){ if(typeof showToast === "function"){ showToast( "ابتدا کلمات را انتخاب کنید." ); } return; } speak( sentence ); }catch(error){ console.error( "Negin AAC current sentence error:", error ); } } /* ========================================================= 🎚 سرعت آهسته ========================================================= */ function setSlowSpeech(){ speechRate = 0.55; speakCurrentSentence(); } /* ========================================================= 🎚 سرعت عادی ========================================================= */ function setNormalSpeech(){ speechRate = 0.85; speakCurrentSentence(); } /* ========================================================= 🎚 سرعت کمی سریع‌تر ========================================================= */ function setFastSpeech(){ speechRate = 1; speakCurrentSentence(); } /* ========================================================= 📱 بررسی صدا هنگام بارگذاری صفحه ========================================================= */ function initializeNeginSpeech(){ if(!speechSupported()){ console.warn( "Negin AAC: Speech synthesis is not supported." ); return; } loadVoices(); /* چند بار بررسی می‌کنیم، چون در Android ممکن است صدا با تأخیر در دسترس قرار بگیرد. */ setTimeout( loadVoices, 300 ); setTimeout( loadVoices, 1000 ); setTimeout( loadVoices, 2000 ); } /* ========================================================= اجرای سیستم صدا ========================================================= */ if( document.readyState === "loading" ){ document.addEventListener( "DOMContentLoaded", initializeNeginSpeech ); }else{ initializeNeginSpeech(); } /* ========================================================= 🧪 اطلاعات تشخیصی صدا ========================================================= */ function diagnoseSpeech(){ console.log( "====================================" ); console.log( "🔊 NEGIN AAC SPEECH DIAGNOSTIC" ); console.log( "====================================" ); console.log( "Speech supported:", speechSupported() ); if(!speechSupported()){ console.log( "❌ Speech synthesis is not supported." ); return; } loadVoices(); console.log( "Available voices:", availableVoices ); console.log( "Persian voice:", selectedPersianVoice ); if(selectedPersianVoice){ console.log( "✅ Persian voice detected:" ); console.log( "Name:", selectedPersianVoice.name ); console.log( "Language:", selectedPersianVoice.lang ); console.log( "Local:", selectedPersianVoice.localService ); }else{ console.warn( "⚠️ No Persian voice detected." ); } console.log( "====================================" ); } /* ========================================================= 🧪 تست کامل سیستم ========================================================= */ function runSpeechTest(){ console.log( "🔊 Starting Negin AAC speech test..." ); if(!speechSupported()){ if(typeof showToast === "function"){ showToast( "مرورگر از قابلیت تبدیل متن به صدا پشتیبانی نمی‌کند." ); } return; } loadVoices(); if(!selectedPersianVoice){ if(typeof showToast === "function"){ showToast( "صدای فارسی پیدا نشد. تنظیمات Text-to-Speech دستگاه را بررسی کنید." ); } diagnoseSpeech(); return; } speak( "سلام. این آزمایش صدای نگین AAC است. صدای فارسی با موفقیت آماده شده است." ); }
